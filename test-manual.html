// Wait for Desmos calculator function findCalculator() { if (window.Calc && typeof window.Calc.getState === 'function') { console.log('✅ Found window.Calc'); return window.Calc; } const containers = document.querySelectorAll('.dcg-calculator-api-container'); for (let container of containers) { if (container.calculator && typeof container.calculator.getState === 'function') { console.log('✅ Found container.calculator'); return container.calculator; } } return null; } // Extract equations function extractEquations() { const calc = findCalculator(); if (!calc) { console.error('❌ Calculator not found'); return null; } const state = calc.getState(); if (!state || !state.expressions || !state.expressions.list) { console.error('❌ Invalid state'); return null; } console.log('📊 Found', state.expressions.list.length, 'expressions'); const equations = []; state.expressions.list.forEach((expr, i) => { if (expr.type === 'folder') { equations.push({ type: 'folder', title: expr.title || 'Folder' }); console.log('[' + i + '] 📁', expr.title); } else if (expr.latex) { equations.push({ latex: expr.latex, color: expr.color }); console.log('[' + i + '] 📐', expr.latex.substring(0, 40)); } }); return equations; } // Run extraction const result = extractEquations(); console.log('✅ Extraction complete:', result); result;
